\begin{frame}{The advisory decision}
\Large A client should be able to inspect what an agent understood before anything changes.\par\vspace{8mm}
\normalsize
\begin{itemize}\item Product comparison provides context for the choice.
\item The profile sets visible allocation limits.
\item Contradictory instructions trigger clarification.
\item A reviewed proposal requires a separate simulation approval.\end{itemize}
\vspace{5mm}\small Our target is an auditable completed workflow, not an unsupported return forecast.
\end{frame}
\begin{frame}{The working client journey}
\begin{enumerate}
\item Compare six fictional sleeves and their principal risks.
\item Complete horizon, liquidity, risk, loss tolerance and experience.
\item Interpret and confirm explicit allocation bounds.
\item Inspect the allocation, risk envelope and constraint residuals.
\item Approve and execute a specific simulated rebalance.
\end{enumerate}
\vspace{6mm}
\textbf{Demo mandate:} TECH at most 15\%, BOND at least 20\%, CASH at least 15\%.\par
\small Optional LLM mode supports prose. Offline mode uses a precise command grammar.
\end{frame}
\begin{frame}{A constrained role for the language model}
\Large The model drafts bounds. The application controls permission.\par\vspace{7mm}
\normalsize\begin{itemize}
\item Only product IDs, inequality directions and numerical weights enter the draft.
\item Unexpected fields, nonfinite values and unsupported products fail validation.
\item The user reviews the interpretation before optimization.
\item Provider failures stop explicitly; the demo remains usable offline.
\end{itemize}
\vspace{5mm}\footnotesize Live provider execution is unverified in this release. Injected responses test the adapter and its failure boundary.
\end{frame}
\begin{frame}{A robust allocation problem we can inspect}
\[
\min_{w\in\mathcal W}\ -\bar r^Tv+\gamma\widehat C_\alpha(v)
+\epsilon\Bigl(1+\frac{\gamma}{1-\alpha}\Bigr)\|v\|_\infty
+k\|w-w_0\|_1
\]
\begin{itemize}
\item Daily units throughout. CASH has zero modeled return.
\item The Wasserstein epigraph objective reduces to a sparse LP.
\item Policy and confirmed client bounds remain hard constraints.
\item Independent checks reconstruct the objective and verify feasibility.
\end{itemize}
\vspace{3mm}\footnotesize Standard robust-optimization tools, integrated into the approval workflow. The uncertainty radius is illustrative, not statistically calibrated.
\end{frame}
\begin{frame}{Conflicting instructions stop the workflow}
\Large ``EQ at most 5\%, and EQ at least 35\%.''\par\vspace{8mm}
\normalsize
\textbf{Result: NEEDS\_REVISION}\par\vspace{4mm}
No portfolio is issued. No constraint is silently relaxed. No approval token is created.\par\vspace{7mm}
The client can revise the visible mandate and request a fresh proposal.
\end{frame}
